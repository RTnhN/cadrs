# SMLib (from NVIDIA's usd-brep, Apache-2.0) built as a test oracle for the native kernel.
#   cmake -S tools/smlib_oracle -B target/smlib_oracle -DUSD_BREP_DIR=<usd-brep checkout> -DCMAKE_BUILD_TYPE=Release
#   cmake --build target/smlib_oracle -j
cmake_minimum_required(VERSION 3.16)
project(smlib_oracle CXX)
set(CMAKE_CXX_STANDARD 17)
set(USD_BREP_DIR "" CACHE PATH "The usd-brep checkout")
if(NOT EXISTS "${USD_BREP_DIR}/source/SMLib/src")
  message(FATAL_ERROR "Set USD_BREP_DIR to a usd-brep checkout")
endif()
set(SMLIB ${USD_BREP_DIR}/source/SMLib)
set(SMAPI ${USD_BREP_DIR}/source/SM_API)
file(GLOB SMLIB_SRC ${SMLIB}/src/*.cpp)
list(FILTER SMLIB_SRC EXCLUDE REGEX "StdAfx.cpp$")
file(GLOB SMAPI_SRC ${SMAPI}/src/*.cpp)
list(FILTER SMAPI_SRC EXCLUDE REGEX "StdAfx.cpp$")
set(WARN -w -Wno-write-strings -fpermissive)
add_library(smlib STATIC ${SMLIB_SRC})
target_include_directories(smlib PUBLIC ${SMLIB}/inc ${CMAKE_CURRENT_SOURCE_DIR}/tbb_shim)
target_compile_definitions(smlib PUBLIC NDEBUG NOMINMAX)
# SMLib's error messages (file and line of each failure, through the error callback): a
# diagnostic build, slower.
option(SMLIB_MESSAGES "Build SMLib with its error messages" OFF)
if(SMLIB_MESSAGES)
  target_compile_definitions(smlib PUBLIC SM_DEBUG_CODE=1)
endif()
target_compile_options(smlib PRIVATE ${WARN})
add_library(smapi STATIC ${SMAPI_SRC})
target_include_directories(smapi PUBLIC ${SMAPI}/inc)
target_link_libraries(smapi PUBLIC smlib)
target_compile_options(smapi PRIVATE ${WARN})
add_executable(smlib_oracle oracle.cpp)
target_link_libraries(smlib_oracle PRIVATE smapi smlib pthread)
target_compile_options(smlib_oracle PRIVATE ${WARN})
